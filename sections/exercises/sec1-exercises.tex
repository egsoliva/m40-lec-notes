\subsection{Exercises}
\begin{problem}
Let $C$ be a matrix and $C = [c_{ij}]_{3 \times 4}$ wherein $c_{ij} = (-1)^{i+j}$. Draw the matrix.
\end{problem}

\begin{problem}
What are the entries $a_{21}$, and $a_{23}$ of the matrix $A = \bmat{1 & 2 & 5 \\ 2 & 7 & 8 \\ 0 & 9 & 4}$.
\end{problem}

\begin{problem}
Find infinitely many matrices $B$ such that $BA = I_2$ when
$$A = \bmat{2 & 3 \\ 1 & 2 \\ 1 & 1}$$
and prove that there is no matrix $C$ such that $AC = I_3$.
\end{problem}

\begin{problem}
What conditions must be satisfied for two square matrices $A, B \in M_n(\RR)$ such that $AB = BA = I_n$?
\end{problem}

\begin{problem}
Find a formula for $\bmat{1 & 1 & 1 \\ 0 & 1 & 1 \\ 0 & 0 & 1}^n$, and prove it by induction.
\end{problem}
