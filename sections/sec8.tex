\section{The Fundamental Subspaces and Rank of a Matrix}

\begin{definition}
Let $A \in M_{m \times n}(\RR)$ with rows $R_1, R_2, \dots, R_m$ and columns $C_1, C_2, \dots, C_n$.
\begin{enumerate}
    \item The \textbf{row space} of $A$ is $R(A) := \operatorname{span}\{R_1, R_2, \dots, R_m\} \leq M_{1 \times n}(\RR)$
    \item The \textbf{column space} of $A$ is $C(A) := \operatorname{span}\{C_1, C_2, \dots, C_n\} \leq M_{m \times 1}(\RR)$
    \item The \textbf{row rank} (respectively \textbf{column rank}) of $A$ is $\dim R(A)$ (respectively $\dim C(A)$)
\end{enumerate}
\end{definition}

\begin{remark}
Applying elementary row operations to $A$ produces a linear combination
of the rows of $A$. If $A$ is row equivalent to $B$, then $R(B) = R(A)$.
\end{remark}

To find bases for $R(A)$ and $C(A)$:
\begin{enumerate}
    \item Use elementary row operations to bring $A$ to a matrix $U$ in REF.
    \item Basis for $C(A) = $ columns of $A$ corresponding to columns of $U$ containing leading entries.
    \item Basis for $R(A) = $ nonzero rows of the REF matrix $U$.
\end{enumerate}

\begin{example}
For each matrix $A$, find bases for $R(A)$ and $C(A)$. Identify the
row and column ranks of $A$.
\begin{enumerate}
    \item $A = \bmat{1 & 0 \\ 4 & -2 \\ 2 & 3}$
    
    \textit{Solution.} 

    $$\bmat{1 & 0 \\ 4 & -2 \\ 2 & 3} \xrightarrow{\text{RREF}} \bmat{1 & 0 \\ 0 & 1 \\ 0 & 0}$$
    Recall that the bases of $R(A)$ are the nonzero rows of the matrix in REF. And 
    the bases for $C(A)$ are the columns of $A$ corresponding to the columns in REF 
    that have leading entries. Thus,
    $$\text{Basis for }R(A) = \{\bmat{1 & 0}, \bmat{0 & 1}\}, \quad \text{Basis for }C(A) = \{\bmat{1 & 4 & 2}\tpse, \bmat{0 & -2 & 3}\tpse\}$$
    $$\therefore \text{row rank of }A = 2 = \text{col rank of } A$$

    \item $A = \bmat{1 & -2 & 3 \\ -3 & 7 & 8 \\ 2 & 0 & -10}$
    
    \textit{Solution.}
    $$\bmat{1 & -2 & 3 \\ -3 & 7 & 8 \\ 2 & 0 & -10} \xrightarrow{\text{RREF}} \bmat{1 & 0 & 5 \\ 0 & 1 & 1 \\ 0 & 0 & 0}$$
    $$\text{Basis for }R(A) = \{\bmat{1 & 0 & 5}, \bmat{0 & 1 & 1}\}, \quad \text{Basis for }C(A) = \{\bmat{1 & -3 & 2}\tpse, \bmat{-2 & 7 & 0}\tpse\}$$
    $$\therefore \text{row rank of }A = 2 = \text{col rank of } A$$

\end{enumerate}
\end{example}

\begin{theorem}
The row rank and column rank of any $A \in M_{m \times n}(\RR)$ are equal.
\end{theorem}

\begin{definition}
The rank of $A \in M_{m \times n}(\RR)$, denoted by $\rank A$, is the common value of the row rank and column ranks of $A$.
\end{definition}

\begin{definition}
Let $A \in M_{m \times n}(\RR)$. The null space of $A$, denoted by $N(A)$, is the solution set of the system $Ax = 0_{m \times 1}$. That is,
$$N(A) = \{x \in M_{n \times 1}(\RR) \mid Ax = 0_{m \times 1}\}$$
\end{definition}

\begin{remark}
If $A \in M_{m \times n}(\RR)$, then $N(A) \leq M_{n \times 1}(\RR)$. The \textbf{nullity} of $A$, denoted by $\upsilon(A)$, is $\dim N(A)$.
\end{remark}

To find $N(A)$, solve for $x$ in the system $Ax = 0_{m \times 1}$.

\begin{example}
Find a basis for $N(A)$ and identify the nullity of $A$.
$$A = \bmat{1 & 0 \\ 4 & -2 \\ 2 & 3}$$
\end{example}

\textit{Solution.} By reducing matrix $A$ to RREF, we get
$$\bmat{1 & 0 \\ 0 & 1 \\ 0 & 0} \longrightarrow 
\begin{cases}
x_1 = 0 \\
x_2 = 0
\end{cases}
$$
Therefore, $N(A) = \left\{\bmat{0 \\ 0}\right\}$ and $\upsilon(A) = 0$.

\begin{example}
Find a basis for $N(A)$ and identify the nullity of $A$.

\textit{Solution.} By reducing the matrix $A$ to RREF, we get
$$\bmat{1 & 0 & 5 \\ 0 & 1 & 1 \\ 0 & 0 & 0} \longrightarrow
\begin{cases}
    x_1 + 5x_3 = 0 \\
    x_2 + x_3 = 0
\end{cases}$$
Let $x_3 = t$, then, $x_1 = -5t, x_2 = -t$,
$$N(A) = \{\bmat{-5t & -t & t}\tpse \mid t \in \RR\} = \spn\{\bmat{-5 & -1 & 1}\tpse\}$$
If $B = \{\bmat{-5 & -1 & 1}\tpse\}$, then $B$ spans $N(A)$ and is linearly independent since 
it consists of one nonzero vector, thus, $B$ is a basis for $N(A)$. Therefore, $\upsilon(A) = 1$.
\end{example}

\begin{example}
Find a basis for the given system of equations
$$
\begin{cases}
x_1 + 2x_3 + x_5 = 0 \\
x_2 + x_3 + x_5 = 0 \\
x_4 - x_5 = 0
\end{cases}
$$

\textit{Solution.} Let us rewrite this in matrix form,
$$\bmat{1 & 0 & 2 & 0 & 1 \\ 0 & 1 & 1 & 0 & 1 \\ 0 & 0 & 0 & 1 & -1}$$
Let $x_3 = s, x_5 = t$, then, $x_1 = -2s - t, x_2 = -s - t, x_4 = t$,
$$N(A) = \{\bmat{-2s - t & -s -t & s & t & t}\tpse \mid s, t \in \RR\}$$
$$=\spn\{\bmat{-2 & -1 & 1 & 0 & 0}\tpse, \bmat{-1 & -1 & 0 & 1 & 1}\tpse\}$$
Thus, $B = \{v_1, v_2\}$ spans $N(A)$ and is linearly independent since $v_1$ is 
not a scalar multiple of $v_2$, therefore, it is a basis for $N(A)$. Therefore, 
$\upsilon(A) = 2$.
\end{example}

\begin{remark}
The nullity of a matrix $A$, $\upsilon(A)$ is the number of columns in any REF 
without leading entries. This is equivalent to the number of free variables 
int $Ax = U_{m \times n}$.
\end{remark}

\begin{theorem}[Rank-Nullity Theorem]
If $A \in M_{m \times n}(\RR)$, then $n = \rank A + \upsilon(A)$,
\end{theorem}

\begin{theorem}
Let $A \in M_n(\RR)$, then the following are equivalent:
\begin{enumerate}
    \item $A$ is nonsingular
    \item $A$ is row equivalent to $I_n$
    \item $A$ is the product of elementary matrices
    \item The system $Ax = b$ has a unique solution for any $b \in M_{n \times 1}(\RR)$
    \item The homogeneous system $Ax = 0_{n \times 1}$ has only the zero solution
    \item $\det A \neq 0$
    \item $\rank A = n$
    \item $\upsilon(A) = 0$
    \item The columns of $A$ form a basis for $M_{n \times 1}(\RR)$
    \item The columns of $A$ are linearly independent
\end{enumerate}
\end{theorem}
